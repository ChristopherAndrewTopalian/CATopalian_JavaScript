AOO Object Methods

// aoo_object_methods_002 

function hi()
{
    console.log('hi');
}

function howdy()
{
    console.log('howdy');
}

function sayThis(whichMessage)
{
    console.log(whichMessage);
}

let people = [
    {
        name: 'Montgomery Scott',
        score: 98,
        action: function() {
            hi();
        }
    },

    {
        name: 'James Kirk',
        score: 93,
        action: function() {
            howdy();
        }
    },

    {
        name: 'Leonard McCoy',
        score: 97,
        action: function() {
            sayThis('Hi Everyone');
        }
    }

];

for (let i = 0; i < people.length; i++)
{
    people[i].action(); 
}

hi
howdy
Hi Everyone
