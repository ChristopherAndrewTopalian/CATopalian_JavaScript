<!DOCTYPE html>
<html lang='en'>
<head>
<title> Paradigm </title>
<meta charset='UTF-8'>
<meta name='viewport' content='width=device-width, initial-scale=1.0'>

<style>

body
{
    background-color: rgb(30, 80, 40); 
    color: rgb(255, 255, 255);
    overflow: hidden;
    margin: 0;
    font-family: sans-serif;
}

</style>

<script>

function hi()
{
    console.log('hi');
}

function howdy()
{
    console.log('howdy');
}

function sayThis(whichMessage)
{
    console.log(whichMessage);
}

let people = [
    {
        name: 'Montgomery Scott',
        score: 98,
        action: function() {
            hi();
        }
    },

    {
        name: 'James Kirk',
        score: 93,
        action: function() {
            howdy();
        }
    },

    {
        name: 'Leonard McCoy',
        score: 97,
        action: function() {
            sayThis('Hi Everyone');
        }
    }

];

function whenLoaded()
{
    for (let i = 0; i < people.length; i++)
    {
        let theBtn = document.createElement('button');
        theBtn.textContent = people[i].name;
        theBtn.onclick = function()
        {
            people[i].action(); 
        };
        document.body.append(theBtn);
    }
}

</script>

</head>

<body onload = 'whenLoaded();'>

</body>

</html>

Description:

A complete, standalone HTML interface demonstrating embedded Action Protocols for the Starfleet crew.
