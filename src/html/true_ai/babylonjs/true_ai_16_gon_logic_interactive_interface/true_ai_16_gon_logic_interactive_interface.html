Babylon.js: True AI 16-Gon Logic Interactive Interface

Building upon the geometric foundation of the True AI 16-Gon, we now activate the interface. By applying Dynamic Textures, we assign the mathematical identity to each logic gate. Furthermore, by utilizing the Action Manager, the vessel becomes interactive, allowing the user to interface directly with the nodes of universal logic. 

// Initialize Engine and Canvas
const canvas = document.getElementById("renderCanvas");
const engine = new BABYLON.Engine(canvas, true, {}, true);

const createScene = function () {
    const scene = new BABYLON.Scene(engine);
    scene.clearColor = new BABYLON.Color4(0.02, 0.02, 0.05, 1); 

    const camera = new BABYLON.ArcRotateCamera("MainCamera", Math.PI / 2, Math.PI / 4, 45, BABYLON.Vector3.Zero(), scene);
    camera.attachControl(canvas, false);

    const light = new BABYLON.HemisphericLight("Starlight", new BABYLON.Vector3(0, 1, 0), scene);
    light.intensity = 0.9;

    const tacticalPlane = BABYLON.MeshBuilder.CreateGround("BasePlane", {width: 40, height: 40, subdivisions: 20}, scene);
    const gridMaterial = new BABYLON.StandardMaterial("GridMat", scene);
    gridMaterial.wireframe = true; 
    gridMaterial.emissiveColor = new BABYLON.Color3(0.1, 0.1, 0.1); 
    tacticalPlane.material = gridMaterial;

    // THE CENTRAL NEURON
    const neuronRadius = 1.5;
    const centerNeuron = BABYLON.MeshBuilder.CreateSphere("CenterNeuron", {diameter: neuronRadius * 2}, scene);
    centerNeuron.position = new BABYLON.Vector3(0, 2, 0); 

    const neuronMat = new BABYLON.StandardMaterial("NeuronMat", scene);
    neuronMat.diffuseColor = new BABYLON.Color3(1, 0.5, 0.8);
    neuronMat.emissiveColor = new BABYLON.Color3(0.8, 0.2, 0.5); 
    centerNeuron.material = neuronMat;

    // THE 16 LOGIC GATES AND NAMES
    const gateCount = 16;
    const radius = 15; 

    // Ordered counter-clockwise starting from the 3 o'clock position
    const gateNames = [
        "RP", "NOR", "CI", "Tautology", 
        "MI", "AND", "LP", "XOR", 
        "RC", "OR", "CNI", "Contradiction", 
        "MNI", "NAND", "LC", "XNOR"
    ];

    const conduitMat = new BABYLON.StandardMaterial("ConduitMat", scene);
    conduitMat.diffuseColor = new BABYLON.Color3(0.8, 0.8, 0.1);
    conduitMat.emissiveColor = new BABYLON.Color3(0.6, 0.6, 0);

    for (let i = 0; i < gateCount; i++) {
        
        let angle = i * ((Math.PI * 2) / gateCount);

        let gateX = Math.cos(angle) * radius;
        let gateZ = Math.sin(angle) * radius;
        let gateY = 2; 

        // Make the box slightly larger to accommodate the text
        let gateBox = BABYLON.MeshBuilder.CreateBox("Gate_" + i, {size: 3}, scene);
        gateBox.position = new BABYLON.Vector3(gateX, gateY, gateZ);
        
        // Rotate the box so the text faces outward
        gateBox.rotation.y = -angle + (Math.PI / 2); 

        // DYNAMIC TEXTURE
        let dynamicTexture = new BABYLON.DynamicTexture("DT_" + i, {width: 256, height: 256}, scene, false);
        
        // Draw the text (Text, X, Y, Font, Text Color, Background Color, InvertY)
        dynamicTexture.drawText(gateNames[i], null, 140, "bold 36px Arial", "white", "#0066cc", true, true);

        let gateMat = new BABYLON.StandardMaterial("GateMat_" + i, scene);
        gateMat.diffuseTexture = dynamicTexture;
        gateMat.emissiveTexture = dynamicTexture; // Makes the text glow!
        gateBox.material = gateMat;

        // INTERACTIVITY (Action Manager)
        gateBox.actionManager = new BABYLON.ActionManager(scene);
        gateBox.actionManager.registerAction(
            new BABYLON.ExecuteCodeAction(BABYLON.ActionManager.OnPickTrigger, function () {
                alert("You have accessed the " + gateNames[i] + " gate.");
            })
        );

        // CONDUITS
        let conduitPath = [
            gateBox.position,
            centerNeuron.position
        ];

        let conduit = BABYLON.MeshBuilder.CreateTube("Conduit_" + i, {
            path: conduitPath,
            radius: 0.15,
            updatable: false
        }, scene);

        conduit.material = conduitMat;
    }

    return scene;
};

const tacticalScene = createScene();

engine.runRenderLoop(function () {
    tacticalScene.render();
});

window.addEventListener("resize", function () {
    engine.resize();
});

Astrometrics Math: Interfaces and Interaction

Dynamic Textures: A 3D box usually requires a pre-made image file to give it details. However, Babylon.js allows us to generate a miniature, invisible 2D HTML Canvas on the fly using BABYLON.DynamicTexture(). We draw our text directly onto this virtual canvas, and Babylon wraps it seamlessly around our 3D logic gate. By applying it to the emissiveTexture, the label generates its own holographic light.

The Array Map: Because our trigonometric loop moves in a perfect, predictable counter-clockwise circle, we can map the exact gate identities by storing them in an array (gateNames). When i = 3, the engine perfectly aligns the position with the string "Tautology".

Action Managers (Interactivity): A 3D scene is just a picture until we give it ears. By attaching a BABYLON.ActionManager to each gate, we tell the engine to listen for an OnPickTrigger (a mouse click or screen tap). Because of JavaScript's block scoping (using let i), each box independently remembers its own name and can alert the user when accessed!
