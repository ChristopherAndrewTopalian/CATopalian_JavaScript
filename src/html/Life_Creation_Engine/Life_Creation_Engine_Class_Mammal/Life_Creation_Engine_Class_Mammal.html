// CLASS DEFINITION
class Mammal {
    constructor(name, color, x, y, speed) {
        this.name = name;
        this.color = color;
        this.x = x;
        this.y = y;
        this.speed = speed;
        this.size = 30;

        // random initial direction
        this.dx = (Math.random() - 0.5) * this.speed;
        this.dy = (Math.random() - 0.5) * this.speed;
        this.directionTimer = 0;
    }

    update(canvasWidth, canvasHeight) {
        // smooth wandering movement
        this.x += this.dx;
        this.y += this.dy;

        // bounce off walls dynamically based on canvas size
        if (this.x < 0 || this.x > canvasWidth) {
            this.dx *= -1;
        }
        if (this.y < 0 || this.y > canvasHeight) {
            this.dy *= -1;
        }

        this.directionTimer++;

        if (this.directionTimer > 60) {
            this.dx = (Math.random() - 0.5) * this.speed;
            this.dy = (Math.random() - 0.5) * this.speed;
            this.directionTimer = 0;
        }
    }

    draw(ctx) {
        // Draw the entity body
        ctx.fillStyle = this.color;
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.size / 2, 0, Math.PI * 2);
        ctx.fill();

        // Draw the perfectly centered name label
        ctx.fillStyle = "white";
        ctx.font = "bold 14px Arial";
        ctx.textAlign = "center";
        ctx.textBaseline = "middle";
        ctx.fillText(this.name, this.x, this.y - this.size);
    }
}

// THE ENGINE ROOM
function gameLoop() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);

    for (let i = 0; i < cats.length; i++) {
        cats[i].update(canvas.width, canvas.height);
        cats[i].draw(ctx);
    }

    requestAnimationFrame(gameLoop);
}

// INITIALIZATION (Optimized for Full-Screen Standalone Deployment)
function whenLoaded() {
    // Deploying our 'ge' shortcut!
    canvas = ge("gameCanvas");
    ctx = canvas.getContext("2d");

    // Get the exact pixel ratio of the users monitor
    const dpr = window.devicePixelRatio || 1;

    // Grab the exact inner dimensions of the browser window
    const STAGE_WIDTH = window.innerWidth;
    const STAGE_HEIGHT = window.innerHeight;

    // Scale the internal canvas memory
    canvas.width = STAGE_WIDTH * dpr;
    canvas.height = STAGE_HEIGHT * dpr;

    // Lock the CSS to the exact same numbers to prevent blurring
    canvas.style.width = STAGE_WIDTH + 'px';
    canvas.style.height = STAGE_HEIGHT + 'px';

    // Scale the drawing context
    ctx.scale(dpr, dpr);

    cats = [
        new Mammal("Mittens", "orange", 100, 200, 2),
        new Mammal("Shadow", "gray", 300, 150, 1.5),
        new Mammal("Snowball", "white", 500, 250, 2.5)
    ];

    gameLoop();
}
