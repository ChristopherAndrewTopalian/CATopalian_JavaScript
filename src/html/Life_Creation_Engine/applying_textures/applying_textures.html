JavaScript Life Creation Engine: Applying Textures

Cadets, a simulation must be visually recognizable to be fully understood. Today, we upgrade from abstract geometric circles to fully textured biological entities. We will load perfectly uniform (square) image files into our engine's memory, and then instruct our Mammal blueprint to paint those images onto the Canvas exactly where our physics math dictates. 

// WORLD VARIABLES & SHORTCUTS
let canvas;
let ctx;
let cats = [];

function ge(whichId)
{
    return document.getElementById(whichId);
}

// TEXTURE CACHING (The Worldwide Image Bank)
const texMittens = new Image(); 
texMittens.src = "../../../../src/media/textures/life/mammals/cats/Mittens.webp";

const texShadow = new Image(); 
texShadow.src = "../../../../src/media/textures/life/mammals/cats/Shadow.webp";

const texSnowball = new Image(); 
texSnowball.src = "../../../../src/media/textures/life/mammals/cats/Snowball.webp";

// THE BLUEPRINT (Physics + Textures)
class Mammal
{
    // Notice we removed 'color' and replaced it with 'textureData'
    constructor(name, textureData, x, y, speed)
    {
        this.name = name;
        this.texture = textureData; 
        this.x = x;
        this.y = y;
        this.speed = speed;
        this.size = 50; 
        
        this.dx = (Math.random() - 0.5) * this.speed;
        this.dy = (Math.random() - 0.5) * this.speed;
        this.directionTimer = 0;
    }

    update(canvasWidth, canvasHeight)
    {
        this.x += this.dx;
        this.y += this.dy;

        if (this.x < 0 || this.x > canvasWidth)
        {
            this.dx *= -1;
        }

        if (this.y < 0 || this.y > canvasHeight)
        {
            this.dy *= -1;
        }

        this.directionTimer++;

        if (this.directionTimer > 60)
        {
            this.dx = (Math.random() - 0.5) * this.speed;
            this.dy = (Math.random() - 0.5) * this.speed;
            this.directionTimer = 0;
        }
    }

    draw(ctx)
    {
        // Asynchronous Safety Check: Only draw if the image has finished downloading!
        if (this.texture.complete && this.texture.naturalWidth !== 0)
        {
            // Draw the texture, centering it on the X and Y coordinates
            ctx.drawImage(this.texture, this.x - this.size/2, this.y - this.size/2, this.size, this.size);
        }
        else
        {
            // Fallback: Draw a gray circle if the image is still loading
            ctx.fillStyle = "gray";
            ctx.beginPath();
            ctx.arc(this.x, this.y, this.size / 2, 0, Math.PI * 2);
            ctx.fill();
        }

        // Draw the name label perfectly centered above the entity
        ctx.fillStyle = "white";
        ctx.font = "bold 14px Arial";
        ctx.textAlign = "center";
        ctx.textBaseline = "middle";
        ctx.fillText(this.name, this.x, this.y - this.size + 5);
    }
}

// THE ENGINE ROOM
function gameLoop()
{
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    
    const logicalWidth = canvas.width / (window.devicePixelRatio || 1);
    const logicalHeight = canvas.height / (window.devicePixelRatio || 1);

    for (let i = 0; i < cats.length; i++)
    {
        cats[i].update(logicalWidth, logicalHeight);
        cats[i].draw(ctx);
    }
    
    requestAnimationFrame(gameLoop);
}

function initFullScreen()
{
    canvas = ge("gameCanvas");
    ctx = canvas.getContext("2d");
    
    const dpr = window.devicePixelRatio || 1;
    const STAGE_WIDTH = window.innerWidth;
    const STAGE_HEIGHT = window.innerHeight;
    
    canvas.width = STAGE_WIDTH * dpr;
    canvas.height = STAGE_HEIGHT * dpr;
    canvas.style.width = STAGE_WIDTH + 'px';
    canvas.style.height = STAGE_HEIGHT + 'px';
    ctx.scale(dpr, dpr);

    // Spawn our Mammals, passing the physical math and the loaded texture!
    cats = [
        new Mammal("Mittens", texMittens, 100, 200, 2),
        new Mammal("Shadow", texShadow, 300, 150, 1.5),
        new Mammal("Snowball", texSnowball, 500, 250, 2.5)
    ];

    gameLoop();
}

Astrometrics Math: URLs and Asynchronous Safety

The Relative Path Trap: Notice the ../../../../src/ at the start of our image source. When your HTML file is buried deep inside a tutorial folder, it does not know where the main src folder is. Every ../ tells the browser to step backward exactly one folder level until it reaches the main directory, at which point it can successfully navigate down into the media files.

Asynchronous Rendering: Hard drives and network servers take time to deliver data. If the Canvas tries to draw a texture using ctx.drawImage() before the file finishes downloading, the entity becomes entirely invisible. By wrapping our draw command in an if (this.texture.complete) statement, the engine safely draws a gray geometric placeholder until the biological texture is fully materialized in memory.
